-- Livraison Manager v18 - numérotation fournisseurs et commandes -- Exécuter UNE SEULE FOIS dans Supabase SQL Editor. alter table public.profiles add column if not exists supplier_code integer; alter table public.colis add column if not exists order_number text; create unique index if not exists profiles_supplier_code_uidx on public.profiles(supplier_code) where supplier_code is not null; create unique index if not exists colis_order_number_uidx on public.colis(order_number) where order_number is not null; -- Attribue automatiquement 1, 2, 3... aux fournisseurs existants sans code. with ranked as ( select id, row_number() over (order by created_at nulls last, id)::integer as rn from public.profiles where role = 'fournisseur' and supplier_code is null ), base as ( select coalesce(max(supplier_code),0) as mx from public.profiles ) update public.profiles p set supplier_code = base.mx + ranked.rn from ranked, base where p.id = ranked.id; create table if not exists public.supplier_order_counters ( fournisseur_id uuid primary key references public.profiles(id) on delete cascade, last_number bigint not null default 0 ); create or replace function public.assign_supplier_code() returns trigger language plpgsql security definer set search_path=public as $$ begin if new.role = 'fournisseur' and new.supplier_code is null then perform pg_advisory_xact_lock(84001); select coalesce(max(supplier_code),0)+1 into new.supplier_code from public.profiles where role='fournisseur'; end if; return new; end $$; drop trigger if exists trg_assign_supplier_code on public.profiles; create trigger trg_assign_supplier_code before insert or update of role on public.profiles for each row execute function public.assign_supplier_code(); create or replace function public.assign_order_number() returns trigger language plpgsql security definer set search_path=public as $$ declare v_prefix integer; v_next bigint; begin if new.order_number is not null then return new; end if; select supplier_code into v_prefix from public.profiles where id=new.fournisseur_id; if v_prefix is null then raise exception 'Code fournisseur manquant'; end if; insert into public.supplier_order_counters(fournisseur_id,last_number) values(new.fournisseur_id,1) on conflict(fournisseur_id) do update set last_number=public.supplier_order_counters.last_number+1 returning last_number into v_next; new.order_number := v_prefix::text || '-' || lpad(v_next::text,6,'0'); return new; end $$; drop trigger if exists trg_assign_order_number on public.colis; create trigger trg_assign_order_number before insert on public.colis for each row execute function public.assign_order_number(); -- Donne aussi un numéro stable aux anciennes commandes qui n'en ont pas. do $$ declare r record; v_prefix integer; v_n bigint; begin for r in select id,fournisseur_id from public.colis where order_number is null order by fournisseur_id,created_at,id loop select supplier_code into v_prefix from public.profiles where id=r.fournisseur_id; insert into public.supplier_order_counters(fournisseur_id,last_number) values(r.fournisseur_id,1) on conflict(fournisseur_id) do update set last_number=public.supplier_order_counters.last_number+1 returning last_number into v_n; update public.colis set order_number=v_prefix::text||'-'||lpad(v_n::text,6,'0') where id=r.id; end loop; end $$; select id, nom, supplier_code from public.profiles where role='fournisseur' order by supplier_code;
